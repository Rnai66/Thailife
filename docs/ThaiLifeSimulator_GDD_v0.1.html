1. วิสัยทัศน์ผลิตภัณฑ์

สร้างเกมจำลองชีวิต 3D ที่มีเอกลักษณ์ประเทศไทย ผู้เล่นสร้างตัวละคร ใช้ชีวิต ทำงาน หาเงิน ปรับปรุงที่อยู่อาศัย สร้างความสัมพันธ์ และพบผู้เล่นคนอื่นในโลกออนไลน์ โดยเริ่มจากต้นแบบขนาดเล็กและขยายอย่างเป็นขั้นตอน

  • เล่นได้ก่อน ขยายทีหลัง
  • เอกลักษณ์ไทยในสถาปัตยกรรม ร้านค้า อาหาร และชุมชน
  • ออกแบบให้เหมาะกับข้อจำกัดมือถือ
  • ข้อมูลสำคัญในระบบออนไลน์ต้องตรวจสอบฝั่งเซิร์ฟเวอร์
  • ขอบเขตงานต้องเหมาะกับผู้พัฒนาคนเดียว

PlayerController.cs

สร้างไฟล์ใน Assets/Scripts/Player/PlayerController.cs

using UnityEngine;

[RequireComponent(typeof(CharacterController))]
public class PlayerController : MonoBehaviour
{
    [SerializeField] private float moveSpeed = 4f;
    [SerializeField] private float gravity = -20f;
    [SerializeField] private Transform cameraTransform;

    private CharacterController controller;
    private float verticalVelocity;

    private void Awake()
    {
        controller = GetComponent<CharacterController>();
    }

    private void Update()
    {
        float x = Input.GetAxisRaw("Horizontal");
        float z = Input.GetAxisRaw("Vertical");

        Vector3 forward = cameraTransform != null
            ? cameraTransform.forward : Vector3.forward;
        Vector3 right = cameraTransform != null
            ? cameraTransform.right : Vector3.right;

        forward.y = 0f;
        right.y = 0f;
        forward.Normalize();
        right.Normalize();

        Vector3 move = Vector3.ClampMagnitude(forward * z + right * x, 1f);

        if (move.sqrMagnitude > 0.001f)
        {
            transform.rotation = Quaternion.Slerp(
                transform.rotation,
                Quaternion.LookRotation(move),
                12f * Time.deltaTime
            );
        }

        if (controller.isGrounded && verticalVelocity < 0f)
            verticalVelocity = -2f;

        verticalVelocity += gravity * Time.deltaTime;
        Vector3 velocity = move * moveSpeed;
        velocity.y = verticalVelocity;
        controller.Move(velocity * Time.deltaTime);
    }
}

CameraFollow.cs

สร้างไฟล์ใน Assets/Scripts/Camera/CameraFollow.cs

using UnityEngine;

public class CameraFollow : MonoBehaviour
{
    [SerializeField] private Transform target;
    [SerializeField] private Vector3 offset = new Vector3(0f, 4f, -6f);
    [SerializeField] private float followSpeed = 8f;
    [SerializeField] private float lookHeight = 1.2f;

    private void LateUpdate()
    {
        if (target == null) return;

        Vector3 desiredPosition = target.position + offset;
        transform.position = Vector3.Lerp(
            transform.position, desiredPosition,
            followSpeed * Time.deltaTime
        );
        transform.LookAt(target.position + Vector3.up * lookHeight);
    }
}

ผูก Component

  1. เพิ่ม PlayerController และ Character Controller ให้ Player
  2. เพิ่ม CameraFollow ให้ Main Camera
  3. ลาก Player ไปใส่ Target ของ CameraFollow
  4. ลาก Main Camera ไปใส่ Camera Transform ของ PlayerController
  5. บันทึก Scene แล้วกด Play

โค้ดเดินใช้ Input Manager แบบเดิม หากโปรเจกต์เปิด Input System แบบใหม่อย่างเดียว ให้ตั้ง Active Input Handling เป็น Both หากมีตัวเลือก หรือเปลี่ยนโค้ดให้ใช้ Input System package โค้ดยังไม่ได้ทดสอบในเครื่องผู้ใช้

7. เกณฑ์ผ่าน Prototype

  • Scene เปิดได้โดยไม่มี compile errors
  • Player เดินและหมุนตัวได้
  • กล้องติดตาม Player
  • Player ไม่ตกทะลุพื้นและชนสิ่งกีดขวางได้
  • Console ไม่มี error สีแดง
  • เพิ่มการควบคุมสัมผัสและทดสอบ iPhone
  • ทดสอบบนอุปกรณ์จริงก่อนเริ่ม Multiplayer

8. ความปลอดภัยและลิขสิทธิ์

  • บันทึกชื่อ asset ผู้สร้าง URL และใบอนุญาต
  • ตรวจสอบสิทธิ์ใช้เชิงพาณิชย์และข้อกำหนดเครดิต
  • ห้ามใช้ asset จากเกมอื่นโดยไม่มีสิทธิ์
  • เมื่อมีระบบออนไลน์ ให้เซิร์ฟเวอร์ตรวจสอบเงินและไอเท็ม
  • อย่าเปิดฐานข้อมูลหรือแผงควบคุมออกอินเทอร์เน็ตโดยไม่มีการป้องกัน
